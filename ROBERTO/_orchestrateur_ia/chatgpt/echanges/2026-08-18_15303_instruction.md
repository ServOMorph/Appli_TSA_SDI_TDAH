# Audit ROBERTO

## 1. Constats

* **Écart principal de format :** `marie_tests_journal.json` contient `{"entries": [...]}`, alors que `ingest_manual_tests.py` attend en entrée `{"manual_test_results": [...]}`. Le fichier fourni ne peut donc **pas être passé directement au script** : il déclenchera `"'manual_test_results' absent ou invalide dans l'export"`.
* **Le format des entrées du journal est cohérent en lui-même** : `id`, `test_id`, `status`, `comment`, `created_at` sont présents sur les 7 entrées.
* **Les statuts actuels sont cohérents** (`ok` / `nok`) et les commentaires sont soit `null`, soit des chaînes.
* **Le script ne valide pas le schéma des entrées importées.** Un résultat sans `id` provoque notamment un `KeyError` sur `result["id"]`; des champs ou valeurs invalides peuvent également être injectés dans le journal.
* **Le journal existant est supposé structurellement valide** : `journal["entries"]` et `entry["id"]` peuvent provoquer des exceptions si le fichier est corrompu ou mal formé.
* **Les erreurs JSON ne sont pas interceptées** (`JSONDecodeError`) : un export invalide produit une traceback plutôt qu'une erreur utilisateur propre.
* **La déduplication par `id` fonctionne** pour les entrées valides et gère aussi les doublons présents dans le même export.
* **Le tri par `created_at` est adapté aux timestamps ISO-8601 homogènes** actuellement utilisés.
* **Écriture non atomique :** le fichier journal est directement réécrit. Une interruption pendant `json.dump()` pourrait laisser le journal partiellement écrit.

## 2. Corrections de code proposées

La correction minimale consiste à **séparer clairement le format d'export du format du journal**, tout en validant les données avant insertion.

```python
"""Fusionne les résultats de tests manuels d'un export JSON de Marie dans le journal projet."""

import json
import sys
from pathlib import Path

JOURNAL_PATH = (
    Path(__file__).resolve().parent.parent
    / "_contexte"
    / "marie_tests_journal.json"
)

REQUIRED_FIELDS = {"id", "test_id", "status", "comment", "created_at"}
VALID_STATUSES = {"ok", "nok"}


def load_json(path: Path) -> dict:
    try:
        with path.open("r", encoding="utf-8") as f:
            data = json.load(f)
    except json.JSONDecodeError as exc:
        raise ValueError(f"JSON invalide dans {path}: {exc}") from exc

    if not isinstance(data, dict):
        raise ValueError(f"Le fichier {path} doit contenir un objet JSON.")

    return data


def load_journal(path: Path) -> dict:
    if not path.exists():
        return {"entries": []}

    journal = load_json(path)
    entries = journal.get("entries")

    if not isinstance(entries, list):
        raise ValueError("Le journal doit contenir une liste 'entries'.")

    for index, entry in enumerate(entries):
        validate_entry(entry, index)

    return journal


def validate_entry(entry: object, index: int) -> None:
    if not isinstance(entry, dict):
        raise ValueError(f"Entrée #{index} invalide : objet attendu.")

    missing = REQUIRED_FIELDS - entry.keys()
    if missing:
        raise ValueError(
            f"Entrée #{index} invalide : champs manquants : "
            f"{', '.join(sorted(missing))}"
        )

    if not isinstance(entry["id"], str) or not entry["id"]:
        raise ValueError(f"Entrée #{index} invalide : 'id' doit être une chaîne.")

    if not isinstance(entry["test_id"], str) or not entry["test_id"]:
        raise ValueError(
            f"Entrée #{index} invalide : 'test_id' doit être une chaîne."
        )

    if entry["status"] not in VALID_STATUSES:
        raise ValueError(
            f"Entrée #{index} invalide : status={entry['status']!r}, "
            f"attendu parmi {sorted(VALID_STATUSES)}."
        )

    if entry["comment"] is not None and not isinstance(entry["comment"], str):
        raise ValueError(
            f"Entrée #{index} invalide : 'comment' doit être null ou une chaîne."
        )

    if not isinstance(entry["created_at"], str) or not entry["created_at"]:
        raise ValueError(
            f"Entrée #{index} invalide : 'created_at' doit être une chaîne."
        )


def main() -> int:
    if len(sys.argv) != 2:
        print(
            "ERREUR: chemin de l'export JSON manquant. "
            "Usage : python ingest_manual_tests.py <export.json>",
            file=sys.stderr,
        )
        return 1

    export_path = Path(sys.argv[1])
    if not export_path.exists():
        print(f"ERREUR: fichier introuvable : {export_path}", file=sys.stderr)
        return 1

    try:
        export_data = load_json(export_path)

        results = export_data.get("manual_test_results")
        if not isinstance(results, list):
            raise ValueError(
                "'manual_test_results' absent ou invalide dans l'export"
            )

        for index, result in enumerate(results):
            validate_entry(result, index)

        journal = load_journal(JOURNAL_PATH)

        known_ids = {entry["id"] for entry in journal["entries"]}

        added = 0
        skipped = 0

        for result in results:
            if result["id"] in known_ids:
                skipped += 1
                continue

            journal["entries"].append(result)
            known_ids.add(result["id"])
            added += 1

        journal["entries"].sort(
            key=lambda entry: (
                entry["created_at"],
                entry["id"],
            )
        )

        JOURNAL_PATH.parent.mkdir(parents=True, exist_ok=True)

        # Écriture temporaire puis remplacement pour éviter
        # de laisser le journal partiellement écrit.
        temp_path = JOURNAL_PATH.with_suffix(".json.tmp")

        with temp_path.open("w", encoding="utf-8") as f:
            json.dump(journal, f, ensure_ascii=False, indent=2)
            f.write("\n")

        temp_path.replace(JOURNAL_PATH)

    except ValueError as exc:
        print(f"ERREUR: {exc}", file=sys.stderr)
        return 1
    except OSError as exc:
        print(f"ERREUR d'accès fichier: {exc}", file=sys.stderr)
        return 1

    print(
        f"Journal mis à jour : {added} ajoutée(s), "
        f"{skipped} déjà présente(s) (ignorée(s))."
    )
    return 0


if __name__ == "__main__":
    raise SystemExit(main())
```

**Point important :** cette correction conserve le contrat actuel du script : l'export source doit être :

```json
{
  "manual_test_results": [
    {
      "id": "...",
      "test_id": "...",
      "status": "ok",
      "comment": null,
      "created_at": "2026-08-18T12:00:00.000Z"
    }
  ]
}
```

et le fichier `_contexte/marie_tests_journal.json` reste :

```json
{
  "entries": [...]
}
```

Il ne faut donc **pas renommer `entries` en `manual_test_results` dans le journal** : ce sont deux rôles différents. Si Claude Code doit pouvoir ré-ingérer directement le journal actuel, il faudrait au contraire ajouter explicitement un mode de lecture `entries`, plutôt que de masquer cet écart de contrat.
